_VM_ROOM_READY_ENTRY {
    VM_ShowNotice("排山倒海")
}

_VM_CARD_CREATED {
	if(lock>0){
		exit;
	}
	card_id = VM_GetLastCreatedCard()
	col = VM_GetProp(card_id,"grid_col")
	card_type = VM_GetProp(card_id,"plant_id")
	current_level = VM_GetProp(card_id,"current_level")
	skill = VM_GetProp(card_id,"skill")
	shape = VM_GetProp(card_id,"shape")
	VM_ShowNoticeDur(card_id,",",col,",",card_type,",",current_level,",",skill,",",shape)
	lock = 1;
	VM_SpawnPlant(card_type,col,-1,shape,current_level,skill)	
}

_VM_FRAME {
	lock = 0;
}
