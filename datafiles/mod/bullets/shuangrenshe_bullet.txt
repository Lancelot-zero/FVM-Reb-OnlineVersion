// 双刃蛇的刀刃弹（mod 子弹）：直线向右飞，命中带斩杀
// 斩杀规则（原版 obj_double_blade_snake_bullet）：
//   目标 hp/maxhp <= exec_th 且是普通敌人 → 直接秒（hp = 0）；
//   精英/Boss（is_boss）→ 改成打 exec_mul 倍伤害；
//   免疫灰烬的敌人（immune_to_ash）→ 不秒杀，按普通伤害结算；
//   其余情况按 damage 正常结算，伤害类型 pierce。
//
// 卡片侧创建方式：
//     VM_SetProp(0, "_mod_pending_bullet_id", "shuangrenshe_bullet")
//     b = VM_CreateInstance("obj_bullet_mod", x + 40, y - 75)
//     VM_SetProp(b, "damage", atk)
//     VM_SetProp(b, "row", grid_row)
//     VM_SetProp(b, "vx", 8) / "vy", 0
//     VM_SetProp(b, "exec_th", 0.25)     // 斩杀阈值
//     VM_SetProp(b, "exec_mul", 10)      // 精英倍率
//
//   ⚠️ 位移由 obj_bullet_mod 做，这里只保证 vx = 8。
//   ⚠️ 「同一敌人只结算一次」靠 mod_step_enter_condition = "cell"（每跨一格进一次 VM）+ 只查
//      子弹所在那一格实现。**不要**用 VM_InstArrayContains/Add 记名单：脚本没法自己创建实例数组，
//      那两个函数在变量不存在时直接返回 -1 什么都不做，去重会完全失效（穿透弹每帧重复扣血）。
//   ⚠️ 原版斩杀会放 obj_mouse_ash_death 处决特效，插件调不到，这里跳过特效只做伤害。

_OBJECT_CFG {
    if (VM_SpriteExists("spr_shuangrenshe_bullet") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_shuangrenshe_bullet_0.png", 10, 64, 56)
        VM_AliasSpritePerm("spr_shuangrenshe_bullet", "tex/spr_shuangrenshe_bullet_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()

    spr = VM_GetProp(self, "sprite_index")
    if (VM_IsUndefined(spr) == 1) {
        spr = 0 - 1
    }
    if (spr == 0 - 1) {
        VM_SetProp(self, "sprite_index", "spr_shuangrenshe_bullet")
    }

    if (VM_IsUndefined(VM_GetProp(self, "vx")) == 1) {
        VM_SetProp(self, "vx", 8)
    }
    if (VM_IsUndefined(VM_GetProp(self, "vy")) == 1) {
        VM_SetProp(self, "vy", 0)
    }
    if (VM_IsUndefined(VM_GetProp(self, "damage")) == 1) {
        VM_SetProp(self, "damage", 0)
    }
    if (VM_IsUndefined(VM_GetProp(self, "exec_th")) == 1) {
        VM_SetProp(self, "exec_th", 0.25)
    }
    if (VM_IsUndefined(VM_GetProp(self, "exec_mul")) == 1) {
        VM_SetProp(self, "exec_mul", 10)
    }
    if (VM_IsUndefined(VM_GetProp(self, "row")) == 1) {
        VM_SetProp(self, "row", VM_GetProp(self, "grid_row"))
    }

    // 只在「跨格」那一帧进 VM：obj_bullet_mod 每帧已经算好 grid_col / prev_grid_col，零成本。
    // 这样每穿过一格只结算一次 —— 正好等于「同一敌人只命中一次」，
    // 不必再自己记名单（原来那套 VM_InstArrayContains/Add 是无效的：脚本没法创建实例数组，
    // 那两个函数在变量不存在时直接返回 -1 什么都不做，导致穿透弹每帧重复扣血）。
    VM_SetProp(self, "mod_step_enter_condition", "cell")
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    r  = VM_GetProp(self, "row")
    bc = VM_GetProp(self, "grid_col")
    dmg = VM_GetProp(self, "damage")
    th  = VM_GetProp(self, "exec_th")
    mul = VM_GetProp(self, "exec_mul")

    n = VM_GetInstancesInRange("srs_hit", r, r, bc, bc, "enemy", "")
    k = 0
    while (k < n) {
        e = VM_ArrayGet("srs_hit", k)
        hp = VM_GetProp(e, "hp")
        if (VM_IsUndefined(hp) == 0) {
            if (hp > 0) {
                mx = VM_GetProp(e, "maxhp")
                ratio = 1
                if (VM_IsUndefined(mx) == 0) {
                    if (mx > 0) {
                        ratio = hp / (mx * 1.0)   // 血比例：除数带小数，否则整数相除直接截成 0
                    }
                }

                // ⚠️ 精英判定用 is_boss：本项目里没有 is_elite 这个属性（help.md / mod敌人.md
                //    登记的身份标记只有 is_boss 和 immune_to_ash）。原来读 is_elite 恒得 undefined
                //    → 精英被当成普通敌人 → 残血时连 Boss 都会被直接秒掉。
                boss = VM_GetProp(e, "is_boss")
                if (VM_IsUndefined(boss) == 1) {
                    boss = 0
                }

                // 免疫灰烬的敌人不吃秒杀（JSON 里 ash_proof → 实例的 immune_to_ash）
                im = VM_GetProp(e, "immune_to_ash")
                if (VM_IsUndefined(im) == 1) {
                    im = 0
                }

                if (ratio <= th) {
                    if (boss == 1) {
                        // 精英/Boss：不秒杀，改成倍率伤害
                        VM_DamageEnemy(e, dmg * mul, "pierce")
                    } else {
                        if (im == 1) {
                            // 免疫灰烬：按普通伤害结算
                            VM_DamageEnemy(e, dmg, "pierce")
                        } else {
                            // 普通：直接秒
                            VM_SetProp(e, "hp", 0)
                        }
                    }
                } else {
                    VM_DamageEnemy(e, dmg, "pierce")
                }
            }
        }
        k = k + 1
    }
}
