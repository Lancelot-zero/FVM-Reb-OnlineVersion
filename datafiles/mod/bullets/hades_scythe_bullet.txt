//     b = VM_CreateInstance("obj_bullet_mod", x, y)

// 轨迹照原版 obj_hades_scythe_bullet：
//     t 每帧 ±4.8，在 0..950 之间往返

//     py = -14 * sqrt(px) * sin(0.006613879270715354 * px)
// VM 只有 VM_Floor / VM_Ceil，没有 sqrt / sin，
// 所以曲线离线按原公式算好存表，运行时线性插值。
// 表：t = i * (950/96)，i = 0..96，共 97 点，插值误差 < 0.15 像素。
//
// 销毁：obj_bullet_mod 的 destroy_timer 属性，0 = 立即销毁，-1 = 不销毁
// 伤害：VM 没有造成伤害的函数，直接改敌人 hp（obj_enemy_parent 中 hp<=0 即进入死亡）
//       伤害是**帧伤** —— 与敌人重叠期间每帧都扣一次，
//       每帧扣 damage / dmg_frames（dmg_frames 在 _OBJECT_CREATE 里设）。

_OBJECT_STEP {
    self = VM_GetCurCard()

    // ---- 推进 t ----
    dirn = VM_GetProp(self, "dirn")
    t = VM_GetProp(self, "t") + 4.8 * dirn

    if (t >= 950) {
        t = 950
        dirn = -1
    }
    if (t <= 0) {
        if (dirn == -1) {
            VM_SetProp(self, "destroy_timer", 0)
            exit
        }
    }
    VM_SetProp(self, "t", t)
    VM_SetProp(self, "dirn", dirn)

    // 前 40 像素透明（照原版）
    if (t <= 40) {
        VM_SetProp(self, "image_alpha", 0)
    } else {
        VM_SetProp(self, "image_alpha", 1)
    }

    // ---- 查表 + 线性插值求纵向偏移 ----
    idx = t / 9.895833333
    i0 = VM_Floor(idx)
    if (i0 < 0) {
        i0 = 0
    }
    if (i0 > 95) {
        i0 = 95
    }
    frac = idx - i0
    y0 = VM_ArrayGet("hsb_curve", i0)
    y1 = VM_ArrayGet("hsb_curve", i0 + 1)
    py = y0 + (y1 - y0) * frac
    if (dirn == -1) {
        py = 0 - py
    }

    VM_SetProp(self, "x", VM_GetProp(self, "start_x") + t)
    VM_SetProp(self, "y", VM_GetProp(self, "start_y") + py)

    // ---- 碰撞：取子弹周围 3x3 格 → 逐个比 x/y 距离 → 每帧扣血 ----
    // 帧伤：把 damage 按 dmg_frames 帧摊平，接触期间每帧都扣一次。
    // 原版是"每颗子弹对每个敌人只结算一次"，改成持续伤害后单帧值要小得多。
    dmg = VM_GetProp(self, "damage") / VM_GetProp(self, "dmg_frames")
    if (dmg < 1) {
        dmg = 1
    }
    // 命中判定半径（像素）；_OBJECT_CREATE 里已设过，这里读是安全的
    rad = VM_GetProp(self, "hit_range")
    if (rad < 1) {
        rad = 90
    }

    bx = VM_GetProp(self, "x")
    by = VM_GetProp(self, "y")
    bc = VM_GetProp(self, "grid_col")
    br = VM_GetProp(self, "grid_row")

    // 收集子弹周围 3x3 格内的敌人。
    // 只取"子弹所在那一格"会漏：子弹在格子边缘时敌人可能在隔壁格，
    // 明明撞上了却因为不同格而不结算。
    n = VM_GetInstancesInRange("hsb_cell", br - 1, br + 1, bc - 1, bc + 1, "enemy", "")

    k = 0
    while (k < n) {
        e = VM_ArrayGet("hsb_cell", k)

        // 算与敌人的 x/y 距离
        dx = VM_GetProp(e, "x") - bx
        if (dx < 0) {
            dx = 0 - dx
        }
        dy = VM_GetProp(e, "y") - by
        if (dy < 0) {
            dy = 0 - dy
        }

        // 帧伤：只要还在重叠范围里就扣，不去重
        if (dx <= rad) {
            if (dy <= rad) {
                VM_SetProp(e, "hp", VM_GetProp(e, "hp") - dmg)
            }
        }
        k = k + 1
    }
}
