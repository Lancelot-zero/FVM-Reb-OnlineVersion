// 冥王战镰的旋转镰刀子弹：沿 ∞ 形轨迹飞行，穿透伤害
//
// 武器侧创建方式（obj_bullet_mod 是通用 mod 子弹对象）：
//     b = VM_CreateInstance("obj_bullet_mod", x, y)
//     VM_SetProp(b, "mod_type", "hades_scythe_bullet")
//     VM_SetProp(b, "damage", 伤害值)
//
// 轨迹：dx_arr / dy_arr 两个数组存"每帧位移"，x/y 每帧加上第 (计数器 mod 长度) 项。
//   数据按原版 ∞ 轨迹离线算好（原版：t 每帧 ±4.8，在 0..950 往返；
//   px = t；py = -14 * sqrt(px) * sin(0.006613879270715354 * px)），一整圈净位移为 0。
//   想改轨迹就直接改这两串数（长度随意，两串一样长即可）。
//
// 销毁：obj_bullet_mod 的 destroy_timer 属性，0 = 立即销毁，-1 = 不销毁
// 伤害：VM 没有造成伤害的函数，直接改敌人 hp（obj_enemy_parent 中 hp<=0 即进入死亡）
//       伤害是**帧伤** —— 与敌人重叠期间每帧都扣一次，
//       每帧扣 damage / dmg_frames（dmg_frames 在 _OBJECT_CREATE 里设）。

_OBJECT_CFG {
    // 统一写内部 spr 名：内置有就直接用；内置没有才把本地图挂到这个名上
    // 子弹：6 帧，原点 (58,58)
    if (VM_SpriteExists("spr_hades_scythe_bullet_4") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_hades_scythe_bullet_4_0.png", 6, 58, 58)
        VM_AliasSpritePerm("spr_hades_scythe_bullet_4", "tex/spr_hades_scythe_bullet_4_0.png")
    }
}

    VM_SetProp(self, "sprite_index", "spr_hades_scythe_bullet_4")
    VM_SetProp(self, "image_speed", 2)
    VM_SetProp(self, "image_xscale", 1.2)
    VM_SetProp(self, "image_yscale", 1.2)
    VM_SetProp(self, "image_alpha", 0)
    // 出场渐显：引擎每帧做（mod_base_tick），不靠 VM Step —— 8 帧从 0 淡到 1，
    // 约等于以前"前 40 像素透明"（那时靠 Step 里的计数器，现在 Step 只在撞到敌人时才进）
    VM_SetProp(self, "mod_alpha_add", 0.125)
    VM_SetProp(self, "mod_countdown", 8)

    // 入口闸门：只在"撞到敌人"那一帧才进 _OBJECT_STEP。
    //   位移由 obj_bullet_mod 按 dx_arr / dy_arr 每帧做，和这个闸门无关 ——
    //   所以子弹在飞的时候一次 VM 都不进，贴着敌人时每帧进（帧伤照旧）。
    VM_SetProp(self, "mod_step_enter_condition", "attack_collision")

    // 一整个循环走完就销毁（和以前"飞完一趟就消失"一致）；想让它一直绕就改成 -1
    //   路径计数器是 obj_bullet_mod 的 bullet_path_i，由对象自己维护，这里不用建
    VM_SetProp(self, "destroy_timer", 396)

    //    格子间距约 107，所以这里超过约 107 之后再加就不起作用了。
    //    要真的更大，得同时把收集范围放宽（见 _OBJECT_STEP）。
    VM_SetProp(self, "hit_range", 130)

    // 取 20 的含义是"接触大约 20 帧的话，整段总伤害 ≈ 原来的单次命中伤害"。

_OBJECT_STEP {
    self = VM_GetCurCard()

    // 这个块现在只在"撞到敌人"那一帧进（见 _OBJECT_CREATE 的闸门）：
    //   位移/渐显都由 obj_bullet_mod 自己做，这里只管结算伤害。

    // ---- 碰撞：取子弹周围 3x3 格 → 逐个比 x/y 距离 → 每帧扣血 ----
    // 帧伤：把 damage 按 dmg_frames 帧摊平，接触期间每帧都扣一次。
    // 原版是"每颗子弹对每个敌人只结算一次"，改成持续伤害后单帧值要小得多。
    dmg = VM_GetProp(self, "damage") / VM_GetProp(self, "dmg_frames")
    if (dmg < 1) {
        dmg = 1
    }
    // 命中判定半径（像素）；_OBJECT_CREATE 里已设过，这里读是安全的
    rad = VM_GetProp(self, "hit_range")
    if (rad < 1) {
        rad = 90
    }

    bx = VM_GetProp(self, "x")
    by = VM_GetProp(self, "y")
    bc = VM_GetProp(self, "grid_col")
    br = VM_GetProp(self, "grid_row")

    // 收集子弹周围 3x3 格内的敌人。
    // 只取"子弹所在那一格"会漏：子弹在格子边缘时敌人可能在隔壁格，
    // 明明撞上了却因为不同格而不结算。
    n = VM_GetInstancesInRange("hsb_cell", br - 1, br + 1, bc - 1, bc + 1, "enemy", "")

    k = 0
    while (k < n) {
        e = VM_ArrayGet("hsb_cell", k)

        // 算与敌人的 x/y 距离
        dx = VM_GetProp(e, "x") - bx
        if (dx < 0) {
            dx = 0 - dx
        }
        dy = VM_GetProp(e, "y") - by
        if (dy < 0) {
            dy = 0 - dy
        }

        // 帧伤：只要还在重叠范围里就扣，不去重
        if (dx <= rad) {
            if (dy <= rad) {
                VM_SetProp(e, "hp", VM_GetProp(e, "hp") - dmg)
            }
        }
        k = k + 1
    }
}
