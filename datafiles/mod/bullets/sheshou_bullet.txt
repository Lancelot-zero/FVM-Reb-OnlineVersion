// 射手蛇的弹（mod 子弹）：向右直线飞；给了 target_row 就一边飞一边朝那一行靠拢
// 行渐变照原版（pierce_bullet 那套）：每帧把 vy 设成"到目标行的差 × 0.15"，
// 靠近到 30 像素内就把 row 切到目标行（之后的命中判定按新行算）。
//
// 卡片侧创建方式：
//     VM_SetProp(0, "_mod_pending_bullet_id", "sheshou_bullet")
//     b = VM_CreateInstance("obj_bullet_mod", x + 40, y - 75)
//     VM_SetProp(b, "damage", atk)
//     VM_SetProp(b, "row", grid_row)
//     VM_SetProp(b, "vx", 8) / "vy", 0
//     VM_SetProp(b, "target_row", 目标行)     // 可选，不给就直飞
//
//   ⚠️ 命中用 VM_DamageEnemy；同一敌人只结算一次。

_OBJECT_CFG {
    if (VM_SpriteExists("spr_sheshou_bullet") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_sheshou_bullet_0.png", 5, 65, 44)
        VM_AliasSpritePerm("spr_sheshou_bullet", "tex/spr_sheshou_bullet_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()

    spr = VM_GetProp(self, "sprite_index")
    if (VM_IsUndefined(spr) == 1) {
        spr = 0 - 1
    }
    if (spr == 0 - 1) {
        VM_SetProp(self, "sprite_index", "spr_sheshou_bullet")
    }

    if (VM_IsUndefined(VM_GetProp(self, "vx")) == 1) {
        VM_SetProp(self, "vx", 8)
    }
    if (VM_IsUndefined(VM_GetProp(self, "vy")) == 1) {
        VM_SetProp(self, "vy", 0)
    }
    if (VM_IsUndefined(VM_GetProp(self, "damage")) == 1) {
        VM_SetProp(self, "damage", 0)
    }
    if (VM_IsUndefined(VM_GetProp(self, "row")) == 1) {
        VM_SetProp(self, "row", VM_GetProp(self, "grid_row"))
    }
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    // 行渐变：有 target_row 才做
    tr = VM_GetProp(self, "target_row")
    if (VM_IsUndefined(tr) == 0) {
        if (tr >= 0) {
            gy = VM_GetProp(0, "grid_offset_y")
            if (VM_IsUndefined(gy) == 1) {
                gy = 0
            }
            gcs = VM_GetProp(0, "grid_cell_size_y")
            if (VM_IsUndefined(gcs) == 1) {
                gcs = 116
            }
            ty = gy + gcs * tr
            cy = VM_GetProp(self, "y")
            dy = ty - cy
            if (dy < 0) {
                dy = 0 - dy
            }
            VM_SetProp(self, "vy", (ty - cy) * 0.15)
            if (dy < 30) {
                VM_SetProp(self, "row", tr)
            }
        }
    }

    r  = VM_GetProp(self, "row")
    bc = VM_GetProp(self, "grid_col")
    dmg = VM_GetProp(self, "damage")

    // 只在「跨格」那一帧结算（上面那段转向要每帧跑，所以不能用 mod_step_enter_condition = "cell"）：
    // 同一格只结算一次 = 同一敌人只命中一次。原来那套 VM_InstArrayContains/Add 是无效的
    //（脚本没法创建实例数组，那两个函数在变量不存在时返回 -1 什么都不做）。
    if (bc != VM_GetProp(self, "prev_grid_col")) {
        n = VM_GetInstancesInRange("sheshou_hit", r, r, bc, bc, "enemy", "")
        k = 0
        while (k < n) {
            e = VM_ArrayGet("sheshou_hit", k)
            if (VM_GetProp(e, "hp") > 0) {
                VM_DamageEnemy(e, dmg, "normal")
            }
            k = k + 1
        }
    }
}
