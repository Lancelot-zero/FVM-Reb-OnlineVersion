
_OBJECT_CFG{
    // 统一写内部 spr 名：内置有就直接用；内置没有才把本地图挂到这个名上
    if (VM_SpriteExists("spr_mod_test_flame") == 0) {
        VM_LoadSpritePerm_Ex("tex/combined_strip.png", 29,30,90)
        VM_AliasSpritePerm("spr_mod_test_flame", "tex/combined_strip.png")
    }
    }

// 测试产火卡：每隔攻击间隔产出一朵火苗（与小火炉同款 obj_flame 逻辑）
_OBJECT_CREATE {
    self = VM_GetCurCard()
    VM_SetProp(self, "cd", 0)

    VM_SetProp(self, "sprite_index", "spr_mod_test_flame")
    VM_SetProp(self, "attack_anim", 18)
    VM_SetProp(self, "idle_anim", 11)
    VM_SetProp(self, "plant_type", "normal")
    VM_SetProp(self, "flame_produce", 25)
    VM_SetProp(self, "first_produce_delay", 60)
    }

_OBJECT_STEP {
    self = VM_GetCurCard()
    cd = VM_GetProp(self, "cd") + 1
    if (cd >= 78) {
        cd = 0
        f = VM_CreateInstance("flame", VM_GetProp(self, "x"), VM_GetProp(self, "y") - 60)
        VM_SetProp(f, "value", 25)
    }
    VM_SetProp(self, "cd", cd)
}
