_OBJECT_STEP {
    self = VM_GetCurCard()
    p = VM_GetProp(self, "mod_point_parent_player")
    fx = VM_GetProp(self, "fx")
    if (VM_IsUndefined(p) == 0) {
        if (VM_IsDestroyed(p) == 0) {
            need = 0
            if (fx == 0) { need = 1 }
            if (VM_IsDestroyed(fx) == 1) { need = 1 }
            if (need == 1) {
                VM_SetProp(0, "_mod_fx_sprite", 2)
                VM_SetProp(0, "_mod_pending_effect_id", "divine_eye_aura")
                fx = VM_CreateInstance("obj_effect_mod", VM_GetProp(p, "x"), VM_GetProp(p, "y"))
                VM_SetProp(0, "_mod_pending_effect_id", "")
                VM_SetProp(fx, "parent_inst", p)
                VM_SetProp(fx, "sprite_index", "spr_master_shield_effect_2")
                VM_SetProp(fx, "frames", 18)
                VM_SetProp(fx, "rel_x", 0 - 15)
                VM_SetProp(fx, "rel_y", 0 - 5)
                VM_SetProp(self, "fx", fx)
            }
        }
    }
    lv = VM_GetProp(self, "mod_point_gem_level")
    if (lv > 15) { lv = 15 }
    t = VM_GetProp(self, "t") + 1
    if (t >= 150) {
        t = 0
        VM_SetProp(self, "cooldown_timer", 150)
        r = VM_GetProp(self, "mod_point_grid_row")
        c = VM_GetProp(self, "mod_point_grid_col")
        n = VM_GetInstancesInRange("dvt_arr", r - 2, r + 2, c - 2, c + 2, "enemy", "")
        k = 0
        while (k < n) {
            e = VM_ArrayGet("dvt_arr", k)
            dmg = VM_Floor(VM_GetProp(e, "maxhp") * VM_ArrayGet("dvt_rate", lv) + VM_ArrayGet("dvt_atk", lv) + 0.5)
            if (dmg > VM_ArrayGet("dvt_max", lv)) { dmg = VM_ArrayGet("dvt_max", lv) }
            VM_DamageEnemy(e, dmg, "normal")
            k = k + 1
        }
    }
    // 每帧把"剩余帧"写给图标 —— 显示和实际触发严格同步（只在触发时写一次会有偏差）
    VM_SetProp(self, "cooldown_timer", 150 - t)
    VM_SetProp(self, "t", t)
}
