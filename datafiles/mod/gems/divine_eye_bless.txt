// 神佑之眼（纯插件复刻）：首次 60 帧后开始产火苗，之后每 cycle×60 帧一次，产量按等级
// 产火是生成火苗【实例】（obj_flame，落到地上等玩家捡），产量写在实例的 value 上
// 等级取 self.mod_point_gem_level（由 obj_player_character 创建宝石时挂上）

// 图标从 mod 目录带过来（注册宝石时就要能解析，所以放在 _OBJECT_CFG 里先加载）
_OBJECT_CFG {
    if (VM_SpriteExists("spr_divine_blessing_gem_icon") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_divine_blessing_gem_icon_0.png", 1, 40, 40)
        VM_AliasSpritePerm("spr_divine_blessing_gem_icon", "tex/spr_divine_blessing_gem_icon_0.png")
    }
    // 战场贴图（45x45、原点居中；按 1.8 放大 ≈81px。背包用的是上面那张 82x82 的 icon）
    if (VM_SpriteExists("spr_divine_blessing_gem") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_divine_blessing_gem_0.png", 1, 22, 22)
        VM_AliasSpritePerm("spr_divine_blessing_gem", "tex/spr_divine_blessing_gem_0.png")
    }
    // 光环贴图：开局加载一次（图在 effects/tex/，所以路径带 ../）
    if (VM_SpriteExists("spr_master_shield_effect_1") == 0) {
        VM_LoadSpritePerm_Ex("../effects/tex/spr_master_shield_effect_1_0.png", 9, 70, 40)
        VM_AliasSpritePerm("spr_master_shield_effect_1", "../effects/tex/spr_master_shield_effect_1_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()
    // 战场贴图 45×45 按 1.8 放大；背包图标是 82×82 那张
    VM_SetProp(self, "sprite_index", "spr_divine_blessing_gem")
    VM_SetProp(self, "image_xscale", 1.8)
    VM_SetProp(self, "image_yscale", 1.8)
    VM_SetProp(self, "t", 0)
    VM_SetProp(self, "first", 0)
    VM_SetProp(self, "fx", 0)      // 光环特效实例（第一次 STEP 里创建）
    VM_SetProp(self, "dbg", 0)
    VM_SetProp(self, "cooldown_timer", 60)   // 预留属性：首发 1 秒，给图标显示冷却用
    // 每档周期（秒，原版 cycle）
    VM_ArrayClear("dvb_cy")
    VM_ArrayADD("dvb_cy", 15)
    VM_ArrayADD("dvb_cy", 15)
    VM_ArrayADD("dvb_cy", 15)
    VM_ArrayADD("dvb_cy", 14)
    VM_ArrayADD("dvb_cy", 14)
    VM_ArrayADD("dvb_cy", 14)
    VM_ArrayADD("dvb_cy", 13)
    VM_ArrayADD("dvb_cy", 13)
    VM_ArrayADD("dvb_cy", 12)
    VM_ArrayADD("dvb_cy", 11)
    VM_ArrayADD("dvb_cy", 10)
    VM_ArrayADD("dvb_cy", 10)
    VM_ArrayADD("dvb_cy", 10)
    VM_ArrayADD("dvb_cy", 10)
    VM_ArrayADD("dvb_cy", 10)
    VM_ArrayADD("dvb_cy", 10)

    // 每档【单朵】产量（一次出 6~13 朵，朵朵一样多；星级越高单朵越大）
    // 单朵 25 → 50 线性递增
    VM_ArrayClear("dvb_fv")
    VM_ArrayADD("dvb_fv", 25)
    VM_ArrayADD("dvb_fv", 27)
    VM_ArrayADD("dvb_fv", 28)
    VM_ArrayADD("dvb_fv", 30)
    VM_ArrayADD("dvb_fv", 32)
    VM_ArrayADD("dvb_fv", 33)
    VM_ArrayADD("dvb_fv", 35)
    VM_ArrayADD("dvb_fv", 37)
    VM_ArrayADD("dvb_fv", 38)
    VM_ArrayADD("dvb_fv", 40)
    VM_ArrayADD("dvb_fv", 42)
    VM_ArrayADD("dvb_fv", 43)
    VM_ArrayADD("dvb_fv", 45)
    VM_ArrayADD("dvb_fv", 47)
    VM_ArrayADD("dvb_fv", 48)
    VM_ArrayADD("dvb_fv", 50)
}

_OBJECT_STEP {
    self = VM_GetCurCard()
    // 光环：拿到角色后创建一次，之后每帧确认还在
    p = VM_GetProp(self, "mod_point_parent_player")
    fx = VM_GetProp(self, "fx")
    db = VM_GetProp(self, "dbg")
    if (db == 0) {  VM_SetProp(self, "dbg", 9) }
    if (VM_IsUndefined(p) == 1) {
        if (db < 9) {
            VM_SetProp(self, "dbg", 9)
        }
    } else {
        if (VM_IsDestroyed(p) == 1) {
            if (db < 9) {
                VM_SetProp(self, "dbg", 9)
            }
        } else {
            need = 0
            if (fx == 0) { need = 1 }
            if (VM_IsDestroyed(fx) == 1) { need = 1 }
            if (need == 1) {
                VM_SetProp(0, "_mod_fx_sprite", 1)
                VM_SetProp(0, "_mod_pending_effect_id", "divine_eye_aura")
                fx = VM_CreateInstance("obj_effect_mod", VM_GetProp(p, "x"), VM_GetProp(p, "y"))
                VM_SetProp(0, "_mod_pending_effect_id", "")
                VM_SetProp(fx, "parent_inst", p)
                VM_SetProp(fx, "sprite_index", "spr_master_shield_effect_1")
                VM_SetProp(fx, "frames", 9)
                VM_SetProp(fx, "rel_x", 0 - 15)
                VM_SetProp(fx, "rel_y", 0 - 30)
                VM_SetProp(self, "fx", fx)
                VM_SetProp(fx, "depth", 9)
                }
        }
    }
    lv = VM_GetProp(self, "mod_point_gem_level")
    if (lv > 15) { lv = 15 }

    // 产火：生成火苗【实例】（和原版产火卡一样，弹一下落到地上等玩家捡）
    // 出生点跟角色走 —— 宝石实例自己停在 UI 坐标上，不能拿来当出生点
    flx = VM_GetProp(self, "x")
    fly = VM_GetProp(self, "y")
    if (VM_IsUndefined(p) == 0) {
        if (VM_IsDestroyed(p) == 0) {
            flx = VM_GetProp(p, "x")
            fly = VM_GetProp(p, "y")
        }
    }

    t = VM_GetProp(self, "t") + 1
    f = VM_GetProp(self, "first")
    fire_now = 0
    if (f == 0) {
        if (t >= 60) {
            t = 0
            f = 1
            fire_now = 1
        }
    } else {
        if (t >= VM_ArrayGet("dvb_cy", lv) * 60) {
            t = 0
            fire_now = 1
        }
    }

    // 一次产几朵：基础 6 朵，星级每高 2 级多 1 朵（等级 0-1 → 6 朵，……，14-15 → 13 朵）
    n = 6
    if (lv >= 2) { n = n + 1 }
    if (lv >= 4) { n = n + 1 }
    if (lv >= 6) { n = n + 1 }
    if (lv >= 8) { n = n + 1 }
    if (lv >= 10) { n = n + 1 }
    if (lv >= 12) { n = n + 1 }
    if (lv >= 14) { n = n + 1 }

    if (fire_now == 1) {
        VM_SetProp(self, "cooldown_timer", VM_ArrayGet("dvb_cy", lv) * 60)
        i = 0
        while (i < n) {
            fl = VM_CreateInstance("obj_flame", flx, fly - 50)
            VM_SetProp(fl, "value", VM_ArrayGet("dvb_fv", lv))
            i = i + 1
        }
    }
    // 每帧把"剩余帧"写给图标 —— 显示和实际触发严格同步（只在触发时写一次会有偏差）
    if (f == 0) {
        VM_SetProp(self, "cooldown_timer", 60 - t)
    } else {
        VM_SetProp(self, "cooldown_timer", VM_ArrayGet("dvb_cy", lv) * 60 - t)
    }
    VM_SetProp(self, "t", t)
    VM_SetProp(self, "first", f)
}
