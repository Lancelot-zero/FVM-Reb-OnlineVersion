// 至尊爱神：本行及上下两行各打一发；每 4 轮多一轮超级弹（首轮就有）
// 计时 / 索敌 / 窗口 / state 全在 obj_card_mod；卡只声明窗口 + 发弹
// ⚠️ 窗口写负数（从一轮末尾倒数），别写死帧数：技能会把 cycle 从 84 压到 48

_OBJECT_CFG {
    if (VM_SpriteExists("spr_love_god_3") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_love_god_3_0.png", 29, 70, 90)
        VM_AliasSpritePerm("spr_love_god_3", "tex/spr_love_god_3_0.png")
    }
    if (VM_SpriteExists("spr_love_god_bullet_3") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_love_god_bullet_3_0.png", 5, 78, 45)
        VM_AliasSpritePerm("spr_love_god_bullet_3", "tex/spr_love_god_bullet_3_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()

    VM_SetProp(self, "sprite_index", "spr_love_god_3")
    VM_SetProp(self, "idle_anim", 15)
    VM_SetProp(self, "attack_anim", 13)
    VM_SetProp(self, "flash_speed", 5)
    VM_SetProp(self, "plant_type", "normal")
    VM_SetProp(self, "state", 0)
    VM_SetProp(self, "super_bullet", 4)

    // 有敌人且命中下面 4 个窗口才进 VM（0 号窗口 = 超级弹）
    VM_SetProp(self, "mod_step_enter_condition", "norm_attack")

    // 索敌范围 [上, 下, 左, 右]：本行 ±1、自己那一格往右
    VM_InstArrayClear(self, "mod_step_enemy_area")
    VM_InstArrayAdd(self, "mod_step_enemy_area", 1)
    VM_InstArrayAdd(self, "mod_step_enemy_area", 1)
    VM_InstArrayAdd(self, "mod_step_enemy_area", 0)
    VM_InstArrayAdd(self, "mod_step_enemy_area", 99)

    f0 = VM_GetProp(self, "flash_speed")
    if (VM_IsUndefined(f0)) {
        f0 = 5
    }

    // 场地常量（一场战斗不变，读一次就够）
    grows = VM_GetProp(0, "grid_rows")
    gy    = VM_GetProp(0, "grid_offset_y")
    gcs   = VM_GetProp(0, "grid_cell_size_y")
    if (VM_IsUndefined(grows)) {
        grows = 6
    }
    if (VM_IsUndefined(gy)) {
        gy = 0
    }
    if (VM_IsUndefined(gcs)) {
        gcs = 116
    }

    // 开火窗口：0 号 = 超级弹，1/2/3 号 = 三次普通齐射
    VM_InstArrayClear(self, "mod_step_enter_arr")
    VM_InstArrayAdd(self, "mod_step_enter_arr", 0 - 7 * f0)
    VM_InstArrayAdd(self, "mod_step_enter_arr", 0 - 5 * f0)
    VM_InstArrayAdd(self, "mod_step_enter_arr", 0 - 3 * f0)
    VM_InstArrayAdd(self, "mod_step_enter_arr", 0 - 1 * f0)
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    // 进到这里 = 有敌人 且 命中某个开火窗口
    wi  = VM_GetProp(self, "mod_step_enter_index")
    atk = VM_GetProp(self, "atk")
    bx  = VM_GetProp(self, "x")
    r   = VM_GetProp(self, "grid_row")

    // 三行的行号与出生点：上一行 r-1、本行 r、下一行 r+1
    // 越界（最上行往上 / 最下行往下）→ 改打本行、出生点从 x+40 收到 x+20
    up = r - 1
    ux = bx + 40
    if (up < 0) {
        up = r
        ux = bx + 20
    }
    dn = r + 1
    dx = bx + 40
    if (dn >= grows) {
        dn = r
        dx = bx + 20
    }

    if (wi == 0) {
        // 0 号窗口 = 超级弹窗口：每轮 +1，攒够 4 次三行各一发穿透弹
        sb = VM_GetProp(self, "super_bullet") + 1
        if (sb >= 4) {
            VM_SetProp(self, "super_bullet", 0)
            VM_BulletScreenAdd_Ex("spr_pierce_bullet", 5, 0, 1.6, 1, ux, gy + gcs * up + 10, 8, 0, 6 * atk / 13, 0 - 1, 0 - 1, "all", "", "", "explosion", 1, 0)
            VM_BulletScreenAdd_Ex("spr_pierce_bullet", 5, 0, 1.6, 1, bx + 40, gy + gcs * r + 10, 8, 0, 6 * atk / 13, 0 - 1, 0 - 1, "all", "", "", "explosion", 1, 0)
            VM_BulletScreenAdd_Ex("spr_pierce_bullet", 5, 0, 1.6, 1, dx, gy + gcs * dn + 10, 8, 0, 6 * atk / 13, 0 - 1, 0 - 1, "all", "", "", "explosion", 1, 0)
        } else {
            VM_SetProp(self, "super_bullet", sb)
        }
    } else {
        // 1~3 号窗口 = 普通齐射：三行各一发（命中一个敌人就消失）
        VM_PlaySound("snd_shot")
        VM_BulletScreenAdd_Ex("spr_love_god_bullet_3", 5, 0, 1.6, 1, ux, gy + gcs * up + 10, 8, 0, atk, 1, 0 - 1, "normal", "", "", "normal", 5, 0)
        VM_BulletScreenAdd_Ex("spr_love_god_bullet_3", 5, 0, 1.6, 1, bx + 40, gy + gcs * r + 10, 8, 0, atk, 1, 0 - 1, "normal", "", "", "normal", 5, 0)
        VM_BulletScreenAdd_Ex("spr_love_god_bullet_3", 5, 0, 1.6, 1, dx, gy + gcs * dn + 10, 8, 0, atk, 1, 0 - 1, "normal", "", "", "normal", 5, 0)
    }
}
