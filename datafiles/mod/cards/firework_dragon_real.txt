// 真·炽焰花火龙（金卡 · 花火龙最终形态）：一次性烟花，放完自己消失
// 按 mod 分支 obj_firework_dragon_real 的 shape 2 移植。
//
//   Create：flame_produce = 75、first_produce_delay = 480（技能表给）、attack_anim = 24、
//           idle_anim = 9、flash_speed 用父对象默认的 6、invincible = true；
//           shape >= 1 时放下就先吐 2 朵价值 75 的火苗（原样抄，值写死 75）
//   Step  ：只走 first_produce == 0 这一支，放完就没了：
//             attack_timer == 480 + 6*fs / +10*fs / +14*fs  → 各吐 4 朵（2 + 二转 2），三轮 12 朵
//             attack_timer == 480 + 24*fs - 1               → 自毁
//   原对象 Step 没有冰冻判定，这里也不加：加了的话万一自毁那一帧正好被冻住，卡片就永远不死。
//
// first_produce_delay 不在父对象 Other_10 的搬运名单里，这里按技能表
// [480,450,420,390,360,300,240,180,60] 用 skill 档位换算；flame_produce 由卡池数据直接给。

_OBJECT_CFG {
    // 统一写内部 spr 名：内置有就直接用；内置没有才把本地图挂到这个名上。
    // 卡面图标：1 帧，原点 (75, 145) —— JSON 的 "sprite" 用它，必须在注册卡片前挂好
    if (VM_SpriteExists("spr_firework_dragon_icon_2") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_firework_dragon_icon_2_0.png", 1, 75, 145)
        VM_AliasSpritePerm("spr_firework_dragon_icon_2", "tex/spr_firework_dragon_icon_2_0.png")
    }

    // 本体：33 帧，原点 (87, 167)
    if (VM_SpriteExists("spr_firework_dragon_2") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_firework_dragon_2_0.png", 33, 87, 167)
        VM_AliasSpritePerm("spr_firework_dragon_2", "tex/spr_firework_dragon_2_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()

    VM_SetProp(self, "sprite_index", "spr_firework_dragon_2")
    VM_SetProp(self, "idle_anim", 9)
    VM_SetProp(self, "attack_anim", 24)
    VM_SetProp(self, "plant_type", "normal")
    VM_SetProp(self, "invincible", 1)

    // 准备时间按技能档位（原版 register_card_skill("firework_dragon_real", "first_produce_delay", ...)）
    sk = VM_GetProp(self, "skill")
    if (VM_IsUndefined(sk)) {
        sk = 0
    }
    fpd = 480
    if (sk == 1) {
        fpd = 450
    }
    if (sk == 2) {
        fpd = 420
    }
    if (sk == 3) {
        fpd = 390
    }
    if (sk == 4) {
        fpd = 360
    }
    if (sk == 5) {
        fpd = 300
    }
    if (sk == 6) {
        fpd = 240
    }
    if (sk == 7) {
        fpd = 180
    }
    if (sk >= 8) {
        fpd = 60
    }
    VM_SetProp(self, "first_produce_delay", fpd)

    VM_SetProp(self, "attack_timer", 0)
    VM_SetProp(self, "first_produce", 0)
    VM_SetProp(self, "state", 0)

    // 最终形态：放下先吐 2 朵价值 75 的火苗（原对象 Create 里 shape >= 1 那一段）
    cx = VM_GetProp(self, "x")
    cy = VM_GetProp(self, "y") - 60
    c1 = VM_CreateInstance("flame", cx, cy)
    VM_SetProp(c1, "value", 75)
    c2 = VM_CreateInstance("flame", cx, cy)
    VM_SetProp(c2, "value", 75)
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    fs = VM_GetProp(self, "flash_speed")
    if (VM_GetProp(self, "is_slowdown") == 1) {
        fs = fs * 2
    }

    fpd = VM_GetProp(self, "first_produce_delay")
    if (VM_IsUndefined(fpd)) {
        fpd = 480
    }

    at = VM_GetProp(self, "attack_timer")
    fp = VM_GetProp(self, "first_produce")
    st = 0
    fire = 0

    if (fp == 0) {
        if (at <= fpd) {
            at = at + 1
        } else if (at <= fpd + 24 * fs) {
            at = at + 1
            st = 1
        }

        // 三轮烟花
        if (at == fpd + 6 * fs) {
            fire = 1
        }
        if (at == fpd + 10 * fs) {
            fire = 1
        }
        if (at == fpd + 14 * fs) {
            fire = 1
        }

        // 放完自毁（原版 instance_destroy，这里把血打到 0 让父对象销毁）
        if (at == fpd + 24 * fs - 1) {
            VM_SetProp(self, "hp", 0)
        }
    }

    if (fire == 1) {
        fpn = VM_GetProp(self, "flame_produce")
        if (VM_IsUndefined(fpn)) {
            fpn = 75
        }
        if (fpn == 0) {
            fpn = 75
        }
        fx = VM_GetProp(self, "x")
        fy = VM_GetProp(self, "y") - 60

        // 每轮 4 朵（最终形态：2 + 二转 2）
        f1 = VM_CreateInstance("flame", fx, fy)
        VM_SetProp(f1, "value", fpn)
        f2 = VM_CreateInstance("flame", fx, fy)
        VM_SetProp(f2, "value", fpn)
        f3 = VM_CreateInstance("flame", fx, fy)
        VM_SetProp(f3, "value", fpn)
        f4 = VM_CreateInstance("flame", fx, fy)
        VM_SetProp(f4, "value", fpn)
    }

    VM_SetProp(self, "attack_timer", at)
    VM_SetProp(self, "first_produce", fp)
    VM_SetProp(self, "state", st)
}
