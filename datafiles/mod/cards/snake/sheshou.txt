// 射手蛇（生肖卡）：一轮内两次开火，中间一路、上下两路追行
// 照 warm_birds 的原始写法：不设 mod_step_enter_condition（默认每帧进 VM），
// 卡自己用 attack_timer 数帧、自己切 state、自己索敌。
//
//   索敌：本行 ±1 行、自己右侧到边界；有敌人才走计时，没敌人计时清零。
//   一轮 cycle 帧内在两个时刻各开一次火（照原版）：
//     第 cycle - 5*flash_speed 帧：中/上/下三路齐射
//     第 cycle - 3*flash_speed 帧：再一轮三路（最终形态）
//   子弹见 mod_bullets/sheshou_bullet.txt，上/下两发带 target_row 做行渐变。
//
//   贴图：本体 spr_sheshou_2 = 28 帧、143x118、原点 (71, 118)

_OBJECT_CFG {
    if (VM_SpriteExists("spr_sheshou_2") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_sheshou_2_0.png", 28, 71, 118)
        VM_AliasSpritePerm("spr_sheshou_2", "tex/spr_sheshou_2_0.png")
    }
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    fs = VM_GetProp(self, "flash_speed")
    if (VM_IsUndefined(fs)) {
        fs = 5
    }
    if (VM_GetProp(self, "is_slowdown") == 1) {
        fs = fs * 2
    }
    cyc = VM_GetProp(self, "cycle")
    if (VM_IsUndefined(cyc)) {
        cyc = 108
    }
    gr = VM_GetProp(self, "grid_row")
    gc = VM_GetProp(self, "grid_col")

    // 索敌：本行 ±1 行、自己右侧
    grs = VM_GetProp(0, "grid_rows")
    if (VM_IsUndefined(grs)) {
        grs = 5
    }
    gcs = VM_GetProp(0, "grid_cols")
    if (VM_IsUndefined(gcs)) {
        gcs = 9
    }
    r1 = gr - 1
    if (r1 < 0) {
        r1 = 0
    }
    r2 = gr + 1
    if (r2 > grs - 1) {
        r2 = grs - 1
    }
    n = VM_GetInstancesInRange("ss_scan", r1, r2, gc, gcs, "enemy", "")

    at = VM_GetProp(self, "mod_tick_cycle")

    if (n <= 0) {
        // 没敌人：回待机（计时由引擎管，不用清）
        VM_SetProp(self, "state", 0)
        exit
    }

    st = 0
    fire1 = 0
    fire2 = 0

    if (at == cyc - 5 * fs) {
        fire1 = 1
        st = 1
    }
    if (at == cyc - 3 * fs) {
        fire2 = 1
        st = 1
    }

    if (fire1 == 1 || fire2 == 1) {
        atk = VM_GetProp(self, "atk")
        bx  = VM_GetProp(self, "x")
        by  = VM_GetProp(self, "y") - 75

        // 中：直飞
        VM_SetProp(0, "_mod_pending_bullet_id", "sheshou_bullet")
        bm = VM_CreateInstance("obj_bullet_mod", bx + 40, by)
        VM_SetProp(bm, "damage", atk)
        VM_SetProp(bm, "row", gr)
        VM_SetProp(bm, "vx", 8)
        VM_SetProp(bm, "vy", 0)

        // 上
        trow = r1
        VM_SetProp(0, "_mod_pending_bullet_id", "sheshou_bullet")
        bu = VM_CreateInstance("obj_bullet_mod", bx + 40, by)
        VM_SetProp(bu, "damage", atk)
        VM_SetProp(bu, "row", gr)
        VM_SetProp(bu, "target_row", trow)
        VM_SetProp(bu, "vx", 8)
        VM_SetProp(bu, "vy", 0)

        // 下
        drow = r2
        VM_SetProp(0, "_mod_pending_bullet_id", "sheshou_bullet")
        bd = VM_CreateInstance("obj_bullet_mod", bx + 40, by)
        VM_SetProp(bd, "damage", atk)
        VM_SetProp(bd, "row", gr)
        VM_SetProp(bd, "target_row", drow)
        VM_SetProp(bd, "vx", 8)
        VM_SetProp(bd, "vy", 0)
    }

    VM_SetProp(self, "state", st)
}
