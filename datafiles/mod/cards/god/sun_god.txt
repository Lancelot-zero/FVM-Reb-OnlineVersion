// 至尊太阳神（金卡 · 阿波罗神使最终形态）：一次生产 12 朵火苗，首次翻倍
// 按 mod 分支 obj_sun_god 的 shape 3 移植。节奏写法跟暖暖鸡不一样，照原样抄：
//
//   Create：flame_produce = 25、attack_anim = 20、idle_anim = 12（最终形态）、flash_speed = 5、
//           first_produce_delay = 60
//   Step  ：判的不是「动画播完才生产」，而是
//             attack_timer >= first_produce_delay - 40 (=20)  → 生产，随后 attack_timer = -40
//             attack_timer >= cycle - 40                     → 生产，随后 attack_timer = -40
//           也就是「放置后 20 帧立刻产一次，之后每 cycle 帧一次」。
//   产量  ：最终形态一次 event_user(1) 吐 12 朵（6 + 一转2 + 二转2 + 终转2）；
//           原版首次那一下还会再调一次 event_user(1)（「首次产量翻倍」）→ 首产 24 朵。
//   原对象没有 is_frozen 判定（暖暖鸡/双子座才有），这里也保持原样不加。

_OBJECT_CFG {
    // 统一写内部 spr 名：内置有就直接用；内置没有才把本地图挂到这个名上。
    // 卡面图标：29 帧，原点 (54, 120) —— JSON 的 "sprite" 用它，必须在注册卡片前挂好
    if (VM_SpriteExists("spr_sun_god_3_icon") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_sun_god_3_icon_0.png", 29, 54, 120)
        VM_AliasSpritePerm("spr_sun_god_3_icon", "tex/spr_sun_god_3_icon_0.png")
    }

    // 本体：29 帧，原点 (66, 147)
    if (VM_SpriteExists("spr_sun_god_3") == 0) {
        VM_LoadSpritePerm_Ex("tex/spr_sun_god_3_0.png", 29, 66, 147)
        VM_AliasSpritePerm("spr_sun_god_3", "tex/spr_sun_god_3_0.png")
    }
}

_OBJECT_CREATE {
    self = VM_GetCurCard()

    VM_SetProp(self, "sprite_index", "spr_sun_god_3")
    // 贴图 29 帧：idle 播 0..idle_anim、攻击从 idle_anim+1 起跳，要求 idle+attack+2 <= 29。
    // 原版 12/20 超了 3 帧，按比例压到 10/17。
    VM_SetProp(self, "idle_anim", 10)
    VM_SetProp(self, "attack_anim", 17)
    VM_SetProp(self, "flash_speed", 5)
    VM_SetProp(self, "plant_type", "normal")

    VM_SetProp(self, "first_produce_delay", 60)
    VM_SetProp(self, "attack_timer", 0)
    VM_SetProp(self, "first_produce", 0)
    VM_SetProp(self, "state", 0)

    // 一轮 1500 帧里只有 2 帧真有事做：第 20 帧开火（首轮 24 朵、之后每轮 12 朵）、第 1461 帧起播攻击动画。
    // 用 "norm" + 窗口表，其余 1498 帧不进 VM。
    VM_InstArrayClear(self, "mod_step_enter_arr")
    VM_InstArrayAdd(self, "mod_step_enter_arr", 20)
    VM_InstArrayAdd(self, "mod_step_enter_arr", 0 - 89)
    VM_SetProp(self, "mod_step_enter_condition", "norm")
}

_OBJECT_STEP {
    self = VM_GetCurCard()

    fs = VM_GetProp(self, "flash_speed")
    if (VM_GetProp(self, "is_slowdown") == 1) {
        fs = fs * 2
    }

    at = VM_GetProp(self, "mod_tick_cycle")
    cyc = VM_GetProp(self, "cycle")
    fp = VM_GetProp(self, "first_produce")
    st = 0
    fire = 0

    // 计时交给引擎：mod_tick_cycle 每帧 +1、到 cycle(1500) 归零，取值 1..1499 / 0。
    // 原来的 attack_timer 是从 -40 起数的错位计时器，换算到 ticker 上：
    //   at = T - 60（T >= 20），at = T + 1440（T <= 19）
    //   开火（at >= cyc-40 = 1460）→ T == 20；攻击动画（at > cyc-20*fs = 1400）→ T >= 1461
    // 所以只需要两个窗口：20（开火，首轮 fire=2）和 1461（切攻击动画）。
    if (at == 20) {
        if (fp == 0) {
            // fire = 2：首次产量翻倍（原版这里 event_user(1) 调两次）
            fire = 2
            fp = 1
        } else {
            fire = 1
        }
    } else if (at >= cyc - 89) {
        st = 1
    }

    if (fire > 0) {
        fpn = VM_GetProp(self, "flame_produce")
        if (VM_IsUndefined(fpn)) {
            fpn = 25
        }
        if (fpn == 0) {
            fpn = 25
        }
        fx = VM_GetProp(self, "x")
        fy = VM_GetProp(self, "y") - 60

        // 一轮 12 朵，fire == 2 时来两轮（首次 24 朵）
        round = 0
        while (round < fire) {
            i = 0
            while (i < 12) {
                fi = VM_CreateInstance("flame", fx, fy)
                VM_SetProp(fi, "value", fpn)
                i = i + 1
            }
            round = round + 1
        }
    }

    VM_SetProp(self, "first_produce", fp)
    VM_SetProp(self, "state", st)
}
